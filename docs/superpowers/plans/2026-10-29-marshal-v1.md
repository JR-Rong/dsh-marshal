# dsh-marshal v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 hub 与页面能观察、续跑、审批和改设置 Codex 与 Claude Code 会话，并把设置变更以 agent 的回读值为准。

**Architecture:** 在 v0 的 node 里引入 `AgentHost`，按 agent 装配适配器。Codex 走一个常驻 `app-server` 子进程（NDJSON JSON-RPC），Claude 走 SDK 的流式输入 `Query`（每活跃会话一个）。两者都把原生事件映射成 v0 的 14 种事件；审批复用 `input.requested`，新增 `decide` 操作。

**Tech Stack:** TypeScript 5.9（纯 ESM，Node ≥ 24）、zod 4、ws 8、vitest 4、`@anthropic-ai/claude-agent-sdk` 0.3.x、Codex app-server 0.153.x

**Docs:** [v1 设计](../specs/2026-10-29-marshal-v1-design.md) · [v1 架构](../specs/2026-10-29-marshal-v1-architecture.md) · [v0 计划](2026-10-29-marshal-v0.md)

> **测试执行位置**：所有测试、端到端与验收在你自己的 checkout、CI 或两台目标机器上执行。本文档不假定任何开发工作站的路径、内网地址、已安装的 agent 或已有运行状态。单元测试全部使用假传输与假 runtime，不要求本机安装 Codex 或 Claude。

```ts
import type { Tunable } from '../adapters/types.ts'

type Edge = readonly [from: string, to: string]

const TRANSITIONS: Readonly<Record<string, readonly Edge[]>> = {
  claude: [
    ['bypassPermissions', 'acceptEdits'],
    ['bypassPermissions', 'default'],
    ['bypassPermissions', 'dontAsk'],
    ['bypassPermissions', 'plan'],
    ['acceptEdits', 'default'],
    ['acceptEdits', 'dontAsk'],
    ['acceptEdits', 'plan'],
    ['auto', 'default'],
    ['auto', 'dontAsk'],
    ['auto', 'plan'],
    ['default', 'dontAsk'],
    ['default', 'plan'],
    ['dontAsk', 'plan'],
  ],
  codex: [
    ['unrestricted', 'workspace'],
    ['unrestricted', 'readonly'],
    ['workspace', 'readonly'],
  ],
}

export interface TightenInput {
  readonly agent: string
  readonly field: 'permission' | 'reasoning'
  readonly from: unknown
  readonly to: unknown
  readonly values?: Tunable['values']
}

/** 只有返回 true 才允许远端改；false 一律按 widen-rejected 处理。 */
export function isTightening(input: TightenInput): boolean {
  if (typeof input.from !== 'string' || typeof input.to !== 'string') return false
  if (input.from === input.to) return false

  const table = TRANSITIONS[input.agent]
  if (table !== undefined) {
    return table.some(([from, to]) => from === input.from && to === input.to)
  }

  const order = input.values?.map(value => value.id) ?? []
  const from = order.indexOf(input.from)
  const to = order.indexOf(input.to)
  return from >= 0 && to >= 0 && to > from
}
```

```ts
import { spawn, type ChildProcess } from 'node:child_process'
import { createInterface } from 'node:readline'

export interface SupervisorOptions {
  readonly name: string
  readonly command: string
  readonly args: readonly string[]
  readonly env?: NodeJS.ProcessEnv
  readonly cwd?: string
  readonly backoffMs?: readonly number[]
  readonly onLine: (line: string) => void
  readonly onExit: (code: number | null, signal: NodeJS.Signals | null) => void
}

export interface Supervisor {
  start(): void
  stop(): Promise<void>
  readonly running: boolean
  readonly restarts: number
}

const DEFAULT_BACKOFF = [1_000, 2_000, 4_000, 8_000, 16_000, 30_000] as const

export function createSupervisor(options: SupervisorOptions): Supervisor {
  const backoff = options.backoffMs ?? DEFAULT_BACKOFF
  const stderr: string[] = []
  let child: ChildProcess | undefined
  let stopped = false
  let restarts = 0
  let timer: NodeJS.Timeout | undefined

  const launch = (): void => {
    if (stopped) return
    const spawned = spawn(options.command, options.args, {
      env: options.env ?? process.env,
      cwd: options.cwd,
      stdio: ['pipe', 'pipe', 'pipe'],
    })
    child = spawned
    const stdout = createInterface({ input: spawned.stdout })
    const errors = createInterface({ input: spawned.stderr })
    stdout.on('line', line => { if (line.length > 0) options.onLine(line) })
    errors.on('line', line => {
      stderr.push(line)
      if (stderr.length > 200) stderr.shift()
    })
    spawned.on('exit', (code, signal) => {
      child = undefined
      options.onExit(code, signal)
      if (stopped) return
      const wait = backoff[Math.min(restarts, backoff.length - 1)] ?? 30_000
      restarts += 1
      timer = setTimeout(launch, wait)
      timer.unref()
    })
  }

  return {
    start() {
      if (child !== undefined) return
      stopped = false
      launch()
    },
    async stop() {
      stopped = true
      if (timer !== undefined) { clearTimeout(timer); timer = undefined }
      const running = child
      if (running === undefined) return
      await new Promise<void>(resolve => {
        const kill = setTimeout(() => { running.kill('SIGKILL'); resolve() }, 2_000)
        running.once('exit', () => { clearTimeout(kill); resolve() })
        running.kill('SIGTERM')
      })
      child = undefined
    },
    get running() { return child !== undefined },
    get restarts() { return restarts },
  }
}
```

`src/agents/host.ts`：

```ts
import type { AgentAdapter, AgentHost, HostLogger } from './types.ts'

export function createAgentHost(adapters: readonly AgentAdapter[], logger: HostLogger): AgentHost {
  const byAgent = new Map(adapters.map(adapter => [adapter.agent, adapter]))
  const pick = (agent: string): AgentAdapter => {
    const adapter = byAgent.get(agent)
    if (adapter === undefined) throw new Error(`agent-missing: ${agent}`)
    return adapter
  }

  return {
    async start() {
      for (const adapter of adapters) {
        try {
          await adapter.start()
          logger.info(`marshal: agent ${adapter.agent} started`)
        } catch (error) {
          logger.warn(`marshal: agent ${adapter.agent} unavailable: ${error instanceof Error ? error.message : 'unknown'}`)
        }
      }
    },
    async stop() {
      await Promise.allSettled(adapters.map(adapter => adapter.stop()))
    },
    adapters() { return adapters },
    async snapshot(signal) {
      const events = (await Promise.all(adapters.map(adapter => adapter.snapshot(signal)))).flat()
      return events.sort((a, b) => {
        const left = (a.payload as { updatedAt?: number }).updatedAt ?? 0
        const right = (b.payload as { updatedAt?: number }).updatedAt ?? 0
        return right - left
      })
    },
    follow(agent, sessionId, signal) { return pick(agent).follow(sessionId, signal) },
    async execute(command) { return pick(command.target.agent).execute(command) },
    describe() {
      return adapters.map(adapter => ({
        agent: adapter.agent,
        probe: (adapter as AgentAdapter & { lastProbe?: import('./types.ts').AgentProbe }).lastProbe
          ?? { available: true, missing: [] },
        capabilities: adapter.capabilities(''),
      }))
    },
  }
}
```

`describe()` 依赖 `lastProbe`，在 Task 6/9 的适配器里赋值。为避免 `as` 断言扩散，在 `AgentAdapter` 上加可选 `lastProbe?: AgentProbe` 更干净——本步骤先加这个可选字段，`host.ts` 直接读 `adapter.lastProbe ?? { available: true, missing: [] }`。

```ts
export interface LineTransport {
  write(text: string): void
  close(): void
}

export interface PeerOptions {
  readonly transport: LineTransport
  readonly onRequest: (method: string, params: unknown, id: number) => Promise<unknown> | undefined
  readonly requestTimeoutMs?: number
  readonly maxFrameBytes?: number
  readonly onProtocolError?: (code: 'invalid-json' | 'frame-too-large') => void
}

export interface CodexPeer {
  request<T = unknown>(method: string, params: unknown): Promise<T>
  notify(method: string, params?: unknown): void
  onNotification(listener: (method: string, params: unknown) => void): () => void
  feed(line: string): void
  dispose(): void
}

interface Pending {
  resolve(value: unknown): void
  reject(error: Error): void
  timer: NodeJS.Timeout
}

const METHOD_NOT_FOUND = -32601
const INTERNAL_ERROR = -32603

export function createCodexPeer(options: PeerOptions): CodexPeer {
  const pending = new Map<number, Pending>()
  const listeners = new Set<(method: string, params: unknown) => void>()
  const limit = options.maxFrameBytes ?? 8_388_608
  let nextId = 1
  let closed = false

  const write = (value: unknown): void => {
    if (closed) return
    options.transport.write(`${JSON.stringify(value)}\n`)
  }

  const handleServerRequest = async (id: number, method: string, params: unknown): Promise<void> => {
    try {
      const result = await options.onRequest(method, params, id)
      if (result === undefined) {
        write({ id, error: { code: METHOD_NOT_FOUND, message: `marshal: unsupported request ${method}` } })
        return
      }
      write({ id, result })
    } catch (error) {
      write({ id, error: { code: INTERNAL_ERROR, message: error instanceof Error ? error.message : 'request-failed' } })
    }
  }

  return {
    request<T>(method: string, params: unknown): Promise<T> {
      if (closed) return Promise.reject(new Error('peer-closed'))
      const id = nextId++
      return new Promise<T>((resolve, reject) => {
        const timer = setTimeout(() => {
          pending.delete(id)
          reject(new Error('agent-timeout'))
        }, options.requestTimeoutMs ?? 30_000)
        timer.unref()
        pending.set(id, { resolve: value => resolve(value as T), reject, timer })
        write({ id, method, params })
      })
    },
    notify(method, params) {
      write(params === undefined ? { method } : { method, params })
    },
    onNotification(listener) {
      listeners.add(listener)
      return () => { listeners.delete(listener) }
    },
    feed(line) {
      if (Buffer.byteLength(line, 'utf8') > limit) {
        options.onProtocolError?.('frame-too-large')
        return
      }
      let message: unknown
      try { message = JSON.parse(line) } catch {
        options.onProtocolError?.('invalid-json')
        return
      }
      if (message === null || typeof message !== 'object') return
      const value = message as { id?: unknown; method?: unknown; params?: unknown; result?: unknown; error?: unknown }
      if (typeof value.id === 'number' && ('result' in value || 'error' in value)) {
        const entry = pending.get(value.id)
        if (entry === undefined) return
        pending.delete(value.id)
        clearTimeout(entry.timer)
        if ('error' in value) {
          const error = value.error as { message?: unknown }
          entry.reject(new Error(typeof error?.message === 'string' ? error.message : 'rpc-error'))
        } else {
          entry.resolve(value.result)
        }
        return
      }
      if (typeof value.method !== 'string') return
      if (typeof value.id === 'number') {
        void handleServerRequest(value.id, value.method, value.params)
        return
      }
      for (const listener of listeners) listener(value.method, value.params)
    },
    dispose() {
      closed = true
      for (const entry of pending.values()) {
        clearTimeout(entry.timer)
        entry.reject(new Error('peer-closed'))
      }
      pending.clear()
      listeners.clear()
      options.transport.close()
    },
  }
}
```

```ts
const toolName = (type: string, value: Record<string, unknown>): string => {
  switch (type) {
    case 'commandExecution': return 'exec'
    case 'fileChange': return 'apply_patch'
    case 'mcpToolCall': return typeof value.tool === 'string' ? `mcp/${value.tool}` : 'mcp'
    case 'dynamicToolCall': return typeof value.tool === 'string' ? value.tool : 'tool'
    case 'webSearch': return 'web_search'
    default: return type
  }
}

export function toMarshalEvents(method: string, params: unknown, frame: Frame): EventFrame[] {
  switch (method) {
    case 'turn/started': {
      const parsed = TurnNotification.parse(params)
      return [frame(parsed.threadId, 'turn.started', { turnId: parsed.turn.id })]
    }
    case 'item/started': {

      const type = String(value.type)
      const toolTypes = new Set(['commandExecution', 'fileChange', 'mcpToolCall', 'dynamicToolCall', 'webSearch'])
      if (!toolTypes.has(type)) return []
      const callId = String(value.id)
      const args: Record<string, unknown> = {}
      if (type === 'commandExecution') {
        if (typeof value.command === 'string') args.command = value.command
        if (typeof value.cwd === 'string') args.cwd = value.cwd
      }
      if (type === 'fileChange' && Array.isArray(value.changes)) args.changes = value.changes.length
      return [frame(threadId, 'tool.started', { callId, name: toolName(type, value), args })]
    }
    case 'item/reasoning/textDelta':
    case 'item/reasoning/summaryTextDelta': {
      const parsed = DeltaNotification.parse(params)
      return [frame(parsed.threadId, 'reasoning.delta', { text: parsed.delta, itemId: parsed.itemId })]

      }
      if (type === 'plan') {

        return [
          frame(threadId, 'tool.started', { callId, name: 'plan' }),
          frame(threadId, 'tool.completed', { callId, name: 'plan', content: text }),
        ]
      }
      if (type === 'reasoning') return []
      return [frame(threadId, 'tool.completed', {

        exitCode: typeof value.exitCode === 'number' ? value.exitCode : undefined,
```

```ts
import type { SettleResult } from '../../adapters/types.ts'
import type { EventFrame } from '../../wire/frames.ts'

export type CodexDecision = 'accept' | 'acceptForSession' | 'decline' | 'cancel'

const DECISIONS = ['accept', 'acceptForSession', 'decline', 'cancel'] as const
const Loose = z.looseObject

export const CommandApprovalParams = Loose({
  threadId: z.string(),
  itemId: z.string(),
  command: z.string().nullish(),
  cwd: z.string().nullish(),
  reason: z.string().nullish(),
  availableDecisions: z.array(z.string()).nullish(),
})

export const FileChangeApprovalParams = Loose({
  threadId: z.string(),
  itemId: z.string(),
  reason: z.string().nullish(),
  grantRoot: z.string().nullish(),
})

  threadId: z.string(),
  itemId: z.string(),
  questions: z.array(Loose({
    id: z.string(),
    header: z.string().nullish(),
    question: z.string(),
    options: z.array(Loose({ label: z.string(), description: z.string().nullish() })).nullish(),
    isSecret: z.boolean().nullish(),
  })),
})

export const PermissionsApprovalParams = Loose({ threadId: z.string() })

export const ElicitationParams = Loose({
  threadId: z.string().nullish(),
  serverName: z.string().nullish(),
  message: z.string().nullish(),
})

export interface CodexApprovalsOptions {
  readonly timeoutMs: number
  readonly emit: (event: EventFrame) => void
  readonly frame: (sessionId: string, kind: EventFrame['kind'], payload: unknown) => EventFrame
}

export interface CodexApprovals {
  handle(method: string, params: unknown): Promise<unknown> | undefined
  settle(threadId: string, requestId: string, decision: string): SettleResult
  answer(threadId: string, requestId: string, answers: unknown): SettleResult
  readonly size: number
  dispose(): void
}

interface Pending {
  readonly threadId: string
  readonly kind: 'approval' | 'question'
  readonly questionIds: readonly string[]
  readonly resolve: (value: unknown) => void
  readonly onTimeout: () => unknown
  readonly timer: NodeJS.Timeout
}

/** 只接受远端传来的 answers 形状：每个问题 id 恰好一次，selected 是字符串数组。 */
export function parseQuestionAnswers(
  value: unknown,
  ids: readonly string[],
): Record<string, { answers: string[] }> | undefined {
  if (typeof value !== 'object' || value === null) return undefined
  const raw = (value as { answers?: unknown }).answers
  if (!Array.isArray(raw) || raw.length !== ids.length) return undefined
  const result: Record<string, { answers: string[] }> = {}
  for (const item of raw) {
    if (typeof item !== 'object' || item === null) return undefined
    const { id, selected } = item as { id?: unknown; selected?: unknown }
    if (typeof id !== 'string' || !ids.includes(id) || result[id] !== undefined) return undefined
    if (!Array.isArray(selected) || !selected.every(label => typeof label === 'string')) return undefined
    result[id] = { answers: selected as string[] }
  }
  return ids.every(id => result[id] !== undefined) ? result : undefined
}

export function createCodexApprovals(options: CodexApprovalsOptions): CodexApprovals {
  const pending = new Map<string, Pending>()
  let counter = 0
  const uniqueId = (base: string): string => `${base}-${++counter}`

  const finish = (threadId: string, requestId: string, outcome: 'remote' | 'local', value: unknown): SettleResult => {
    const entry = pending.get(requestId)
    if (entry === undefined) return 'unknown'
    if (entry.threadId !== threadId) return 'session-mismatch'
    clearTimeout(entry.timer)
    pending.delete(requestId)
    options.emit(options.frame(threadId, 'input.resolved', { requestId, outcome }))
    entry.resolve(value)
    return 'ok'
  }

  const ask = (
    threadId: string,
    requestId: string,
    kind: 'approval' | 'question',
    questionIds: readonly string[],
    card: Record<string, unknown>,
    onTimeout: () => unknown,
  ): Promise<unknown> => new Promise(resolve => {
    const timer = setTimeout(() => {
      pending.delete(requestId)
      options.emit(options.frame(threadId, 'input.resolved', { requestId, outcome: 'timeout' }))
      resolve(onTimeout())
    }, options.timeoutMs)
    pending.set(requestId, { threadId, kind, questionIds, resolve, onTimeout, timer })
    options.emit(options.frame(threadId, 'input.requested', {
      requestId, kind, agent: 'codex', expiresAt: Date.now() + options.timeoutMs, ...card,
    }))
  })

  const decisionsOf = (available: readonly string[] | null | undefined): readonly string[] => {
    if (available === null || available === undefined || available.length === 0) return DECISIONS
    return DECISIONS.filter(decision => available.includes(decision))
  }

  return {
    handle(method, raw) {
      switch (method) {
        case 'item/commandExecution/requestApproval': {
          const params = CommandApprovalParams.parse(raw)
          const detail = [params.command, params.cwd === undefined ? undefined : `cwd: ${params.cwd}`, params.reason]
            .filter((line): line is string => typeof line === 'string' && line.length > 0)
            .join('\n')
          return ask(params.threadId, uniqueId(`a-${params.itemId}`), 'approval', [], {
            title: '运行命令',
            ...(detail.length === 0 ? {} : { detail }),
            decisions: decisionsOf(params.availableDecisions),
          }, () => ({ decision: 'decline' }))
        }
        case 'item/fileChange/requestApproval': {
          const params = FileChangeApprovalParams.parse(raw)
          return ask(params.threadId, uniqueId(`a-${params.itemId}`), 'approval', [], {
            title: '写入文件',
            ...(params.reason === null || params.reason === undefined ? {} : { detail: params.reason }),
            ...(params.grantRoot === null || params.grantRoot === undefined ? {} : { grantRoot: params.grantRoot }),
            decisions: DECISIONS,
          }, () => ({ decision: 'decline' }))
        }
        case 'item/permissions/requestApproval':
          PermissionsApprovalParams.parse(raw)
          return Promise.resolve({ permissions: {}, scope: 'turn' })
        case 'mcpServer/elicitation/request': {
          const params = ElicitationParams.parse(raw)
          if (typeof params.threadId === 'string') {
            const requestId = uniqueId(`e-${params.serverName ?? 'mcp'}`)
            options.emit(options.frame(params.threadId, 'input.requested', {
              requestId, kind: 'approval', agent: 'codex', notifyOnly: true, decisions: [],
              title: 'MCP 请求（远端只提示）',
              ...(params.message === null || params.message === undefined ? {} : { detail: params.message }),
            }))
            options.emit(options.frame(params.threadId, 'input.resolved', { requestId, outcome: 'local' }))
          }
          return Promise.resolve({ action: 'decline', content: null, _meta: null })
        }
        case 'item/tool/requestUserInput': {
          const params = UserInputParams.parse(raw)
          if (params.questions.some(question => question.isSecret === true)) {
            const requestId = uniqueId(`q-${params.itemId}`)
            options.emit(options.frame(params.threadId, 'input.requested', {
              requestId, kind: 'question', agent: 'codex', secret: true, questions: [],
              title: '敏感提问未转发',
            }))
            options.emit(options.frame(params.threadId, 'input.resolved', { requestId, outcome: 'local' }))
            return Promise.resolve({ answers: {} })
          }
          return ask(
            params.threadId,
            uniqueId(`q-${params.itemId}`),
            'question',
            params.questions.map(question => question.id),
            {
              title: '需要你的输入',
              questions: params.questions.map(question => ({
                id: question.id,
                header: question.header ?? undefined,
                question: question.question,
                options: question.options ?? [],
                multiSelect: false,
              })),
            },
            () => ({ answers: {} }),
          )
        }
        default:
          return undefined
      }
    },
    settle(threadId, requestId, decision) {
      const entry = pending.get(requestId)
      if (entry === undefined) return 'unknown'
      if (entry.threadId !== threadId) return 'session-mismatch'
      if (entry.kind !== 'approval') return 'invalid'
      if (!(DECISIONS as readonly string[]).includes(decision)) return 'invalid'
      return finish(threadId, requestId, 'remote', { decision })
    },
    answer(threadId, requestId, answers) {
      const entry = pending.get(requestId)
      if (entry === undefined) return 'unknown'
      if (entry.threadId !== threadId) return 'session-mismatch'
      if (entry.kind !== 'question') return 'invalid'
      const parsed = parseQuestionAnswers(answers, entry.questionIds)
      if (parsed === undefined) return 'invalid'
      return finish(threadId, requestId, 'remote', { answers: parsed })
    },
    get size() { return pending.size },
    dispose() {
      for (const [requestId, entry] of pending) {
        clearTimeout(entry.timer)
        pending.delete(requestId)
        entry.resolve(entry.onTimeout())
      }
    },
  }
}
```

```ts
export interface EventQueue<T> extends AsyncIterable<T> {
  push(value: T): void
  close(): void
}

export function createEventQueue<T>(): EventQueue<T> {
  const values: T[] = []
  let closed = false
  let wake: (() => void) | undefined

  return {
    push(value) {
      if (closed) return
      values.push(value)
      const resume = wake
      wake = undefined
      resume?.()
    },
    close() {
      closed = true
      const resume = wake
      wake = undefined
      resume?.()
    },
    async *[Symbol.asyncIterator]() {
      while (true) {
        const value = values.shift()
        if (value !== undefined) { yield value; continue }
        if (closed) return
        await new Promise<void>(resolve => { wake = resolve })
      }
    },
  }
}
```

```ts
  return mode === 'plan' ? 'on' : 'off'

const message = (error: unknown): string => error instanceof Error ? error.message : String(error)

export function createCodexAdapter(options: CodexAdapterOptions): AgentAdapter {
  const { frame, peer, approvals } = options
  const states = new Map<string, SessionState>()
  const ownership = new Map<string, Ownership>()
  const resumed = new Map<string, Promise<void>>()
  const modelCatalog: CodexModel[] = []
  let lastProbe: AgentProbe | undefined
  let subscribed = false

  const state = (sessionId: string): SessionState => {
    const existing = states.get(sessionId)
    if (existing !== undefined) return existing
    const created: SessionState = { waiters: new Set() }
    states.set(sessionId, created)
    return created
  }

  const tunables = (sessionId: string): readonly Tunable[] => {
    const current = state(sessionId)
    const settings = current.settings
    const model = modelCatalog.find(entry => entry.model === (settings?.model ?? ''))
    const efforts = model?.supportedReasoningEfforts ?? []
    return [
      {
        id: 'model', scope: 'next-turn', effect: 'immediate', editable: true,
        values: modelCatalog.map(entry => ({ id: entry.model, label: entry.displayName ?? entry.model })),
        ...(settings?.model === null || settings?.model === undefined ? {} : { current: settings.model }),
      },
      {
        id: 'reasoning', scope: 'next-turn', effect: 'immediate', editable: true, direction: 'tighten-only',
        values: efforts.map(effort => ({ id: effort.reasoningEffort, label: effort.reasoningEffort })),
        ...(settings?.effort === null || settings?.effort === undefined ? {} : { current: settings.effort }),
      },
      {
        id: 'permission', scope: 'next-turn', effect: 'immediate', editable: true, direction: 'tighten-only',
        values: PERMISSION_VALUES.map(value => ({ ...value })),
        ...(permissionOf(settings) === undefined ? {} : { current: permissionOf(settings) }),
      },
      {
        id: 'plan', scope: 'next-turn', effect: 'immediate', editable: true,
        values: [{ id: 'on', label: '计划模式' }, { id: 'off', label: '普通模式' }],
        current: planOf(settings),
      },
      {
        id: 'goal', scope: 'next-turn', effect: 'immediate', editable: true,
        ...(current.goal === undefined ? {} : { current: current.goal.objective }),
      },
      {
        id: 'skill', scope: 'host', effect: 'needs-restart', editable: false,
        reason: 'Codex 的技能在启动时加载，远端只能查看',
      },
      {
        id: 'plugin', scope: 'host', effect: 'needs-restart', editable: false,
        reason: '插件作用域是宿主，改动需要重启 Codex',
      },
    ]
  }

  const emitCapability = (sessionId: string): void => {
    options.emit(frame(sessionId, 'capability.changed', {
      tunables: tunables(sessionId),
      readbackAt: Date.now(),
    }))
  }

  const subscribe = (): void => {
    if (subscribed) return
    subscribed = true
    peer.onNotification((method, params) => {
      if (method !== 'thread/settings/updated') return
      const parsed = ThreadSettingsUpdated.safeParse(params)
      if (!parsed.success) return
      const target = state(parsed.data.threadId)
      target.settings = parsed.data.threadSettings
      for (const waiter of target.waiters) waiter(parsed.data.threadSettings)
      target.waiters.clear()
      emitCapability(parsed.data.threadId)
    })
  }

  const waitForSettings = (sessionId: string): Promise<ThreadSettings> =>
    new Promise((resolve, reject) => {
      const target = state(sessionId)
      const timer = setTimeout(() => {
        target.waiters.delete(waiter)
        reject(new Error('readback-timeout'))
      }, options.readbackTimeoutMs ?? 5_000)
      const waiter = (settings: ThreadSettings): void => { clearTimeout(timer); resolve(settings) }
      target.waiters.add(waiter)
    })

  const ensure = (sessionId: string): Promise<void> => {
    const existing = resumed.get(sessionId)
    if (existing !== undefined) return existing
    const pending = (async () => {
      const parsed = ThreadResumeResponse.parse(await peer.request('thread/resume', {
        threadId: sessionId,
        excludeTurns: true,
        initialTurnsPage: { limit: 20 },
      }))
      if (parsed.threadSettings !== undefined) state(sessionId).settings = parsed.threadSettings
      ownership.set(sessionId, 'ours')
      try {
        const goal = await peer.request('thread/goal/get', { threadId: sessionId }) as { goal?: { objective?: unknown } }
        if (typeof goal.goal?.objective === 'string') state(sessionId).goal = { objective: goal.goal.objective }
      } catch { /* goal 是可选能力，读不到不影响续跑 */ }
    })().catch(error => {
      resumed.delete(sessionId)
      throw new Error(`resume-failed: ${message(error)}`)
    })
    resumed.set(sessionId, pending)
    return pending
  }

  const verify = (sessionId: string, field: Tunable['id'], desired: unknown, next: ThreadSettings): void => {
    const actual = field === 'model'
      ? next.model
      : field === 'reasoning'
        ? next.effort
        : field === 'permission'
          ? permissionOf(next)
          : field === 'plan'
            ? planOf(next)
            : desired
    if (actual !== desired) {
      throw new Error(`readback-mismatch: ${field} ${String(actual)} != ${String(desired)}`)
    }
    state(sessionId).settings = next
  }

  const configure = async (sessionId: string, args: Record<string, unknown>): Promise<unknown> => {
    const field = args.field as Tunable['id'] | undefined
    if (field === undefined) throw new Error('invalid-args: field is required')
    const tunable = tunables(sessionId).find(item => item.id === field)
    if (tunable === undefined) throw new Error(`unsupported: ${field}`)
    if (!tunable.editable) throw new Error(`not-editable: ${field}`)
    await ensure(sessionId)

    if (field === 'permission' || field === 'reasoning') {
      const from = tunable.current
      if (from === undefined) throw new Error(`readback-timeout: ${field}`)
      if (from === args.value) return { unchanged: true, settings: state(sessionId).settings ?? {} }
      if (!isTightening({ agent: 'codex', field, from, to: args.value, values: tunable.values })) {
        throw new Error(`widen-rejected: ${String(from)} -> ${String(args.value)}`)
      }
    }

    if (field === 'goal') {
      if (args.value === null || args.value === '' || args.value === 'clear') {
        await peer.request('thread/goal/clear', { threadId: sessionId })
        delete state(sessionId).goal
        emitCapability(sessionId)
        return { goal: null }
      }
      if (typeof args.value !== 'string' || args.value.trim() === '') {
        throw new Error('invalid-args: goal requires a non-empty string or clear')
      }
      await peer.request('thread/goal/set', { threadId: sessionId, objective: args.value })
      state(sessionId).goal = { objective: args.value }
      emitCapability(sessionId)
      return { goal: state(sessionId).goal }
    }

    if (field === 'model' && modelCatalog.length > 0 && !modelCatalog.some(entry => entry.model === args.value)) {
      throw new Error(`invalid-args: unknown model ${String(args.value)}`)
    }

    const params: Record<string, unknown> = { threadId: sessionId }
    if (field === 'model') params.model = args.value
    if (field === 'reasoning') params.effort = args.value
    if (field === 'permission') {
      const mapped = PERMISSION_PARAMS[String(args.value)]
      if (mapped === undefined) throw new Error(`invalid-args: unknown permission ${String(args.value)}`)
      Object.assign(params, mapped)
    }
    if (field === 'plan') {
      const settings = state(sessionId).settings
      if (settings === undefined) throw new Error('unsupported: plan requires settings readback')
      const mode = args.value === true || args.value === 'on' || args.value === 'plan' ? 'plan' : 'default'
      params.collaborationMode = { mode, settings }
      args = { ...args, value: mode }
    }

    const response = ThreadSettingsUpdateResponse.parse(await peer.request('thread/settings/update', params))
    const next = response.threadSettings ?? response.settings ?? await waitForSettings(sessionId)
    verify(sessionId, field, args.value, next)
    emitCapability(sessionId)
    return { settings: next }
  }

  return {
    agent: 'codex',
    get lastProbe() { return lastProbe },
    capabilities(sessionId) {
      const capability: Capability = {
        observe: 'stream',
        dispatch: 'prompt',
        reasoning: true,
        toolDetail: 'args+output',
        promptFidelity: 'native',
        tunables: tunables(sessionId),
      }
      return capability
    },
    frame,
    tunables,
    ownership(sessionId) { return ownership.get(sessionId) ?? 'foreign-idle' },
    async probe() { return lastProbe ?? { available: false, missing: ['codex'], reason: 'not started' } },
    async start() {
      try {
        await peer.request('initialize', {
          clientInfo: { name: 'dsh-marshal', version: options.version ?? '0' },
          capabilities: { experimentalApi: true, requestAttestation: false },
        })
        peer.notify('initialized')
        lastProbe = { available: true, ...(options.version === undefined ? {} : { version: options.version }), missing: [] }
      } catch (error) {
        lastProbe = { available: false, missing: ['codex'], reason: message(error) }
        return
      }
      subscribe()
      try {
        const catalog = ModelListResponse.parse(await peer.request('model/list', {}))
        modelCatalog.splice(0, modelCatalog.length, ...catalog.data)
      } catch { /* 目录拿不到时仍可观察与提问，只是不能远改模型 */ }
    },
    async stop() {
      resumed.clear()
      for (const target of states.values()) for (const waiter of target.waiters) waiter(target.settings ?? {})
      states.clear()
      ownership.clear()
      peer.dispose()
    },
    async snapshot(signal) {
      const threads = await listThreads(peer)
      if (signal.aborted) return []
      const events = toSessionEvents(threads, frame)
      const patched = events.map(event => ownership.get(event.sessionId) === 'ours'
        ? { ...event, payload: { ...(event.payload as object), ownership: 'ours' as const } }
        : event)
      for (const thread of threads) {
        patched.push(frame(thread.id, 'capability.changed', {
          tunables: tunables(thread.id),
          readbackAt: Date.now(),
        }))
      }
      return patched
    },
    async *follow(sessionId, signal) {
      const queue: EventQueue<EventFrame> = createEventQueue()
      const off = peer.onNotification((method, params) => {
        const parsed = ThreadIdOnly.safeParse(params)
        if (!parsed.success || parsed.data.threadId !== sessionId) return
        for (const event of toMarshalEvents(method, params, frame)) queue.push(event)
      })
      const onAbort = (): void => { off(); queue.close() }
      signal.addEventListener('abort', onAbort, { once: true })
      try {
        await ensure(sessionId)
        queue.push(frame(sessionId, 'session.updated', {
          ownership: 'ours', running: true, updatedAt: Date.now(),
        }))
        yield* queue
      } finally {
        off()
        signal.removeEventListener('abort', onAbort)
      }
    },
    async execute(command) {
      const sessionId = command.target.sessionId
      const args = (command.args ?? {}) as Record<string, unknown>
      switch (command.op) {
        case 'prompt': {
          if (typeof args.text !== 'string' || args.text.trim() === '') {
            throw new Error('invalid-args: text is required')
          }
          await ensure(sessionId)
          const settings = state(sessionId).settings
          const result = await peer.request('turn/start', {
            threadId: sessionId,
            input: [{ type: 'text', text: args.text, text_elements: [] }],
            ...(settings?.model === null || settings?.model === undefined ? {} : { model: settings.model }),
            ...(settings?.effort === null || settings?.effort === undefined ? {} : { effort: settings.effort }),
            ...(settings?.approvalPolicy === null || settings?.approvalPolicy === undefined
              ? {} : { approvalPolicy: settings.approvalPolicy }),
          }) as { turn?: unknown }
          return { accepted: true, turn: result.turn }
        }
        case 'interrupt':
          await peer.request('turn/interrupt', { threadId: sessionId })
          return { interrupted: true }
        case 'decide': {
          if (typeof args.requestId !== 'string' || typeof args.decision !== 'string') {
            throw new Error('invalid-args: requestId and decision are required')
          }
          const outcome = approvals.settle(sessionId, args.requestId, args.decision)
          if (outcome !== 'ok') throw new Error(`${outcome}: ${args.requestId}`)
          return { accepted: true }
        }
        case 'answer': {
          if (typeof args.requestId !== 'string' || args.answers === undefined) {
            throw new Error('invalid-args: requestId and answers are required')
          }
          const outcome = approvals.answer(sessionId, args.requestId, args.answers)
          if (outcome !== 'ok') throw new Error(`${outcome}: ${args.requestId}`)
          return { accepted: true }
        }
        case 'configure':
          return configure(sessionId, args)
        default:
          throw new Error(`unsupported: ${command.op}`)
      }
    },
  }
}
```

```ts
import type { EventFrame } from '../../wire/frames.ts'

type Frame = (sessionId: string, kind: EventFrame['kind'], payload: unknown) => EventFrame

const record = (value: unknown): Record<string, unknown> | undefined =>
  typeof value === 'object' && value !== null ? value as Record<string, unknown> : undefined

/** 只在 `includePartialMessages: true` 下使用：正文与工具起点来自 stream_event。 */
export function toMarshalEvents(message: unknown, sessionId: string, frame: Frame): EventFrame[] {
  const value = record(message)
  if (value === undefined) return []
  switch (value.type) {
    case 'system':
      return system(value, sessionId, frame)
    case 'stream_event':
      return stream(value, sessionId, frame)
    case 'assistant':
      return assistant(value, sessionId, frame)
    case 'user':
      return user(value, sessionId, frame)
    case 'result':
      return [frame(sessionId, 'turn.completed', {
        turnId: String(value.num_turns ?? 0),
        status: value.is_error === true ? 'error' : String(value.subtype ?? 'completed'),
        durationMs: typeof value.duration_ms === 'number' ? value.duration_ms : undefined,
      })]
    default:
      return []
  }
}

function system(value: Record<string, unknown>, sessionId: string, frame: Frame): EventFrame[] {
  if (value.subtype === 'init') {
    return [frame(sessionId, 'session.updated', {
      running: true,
      model: value.model,
      permission: value.permissionMode,
      cwd: value.cwd,
      ownership: 'ours',
      updatedAt: Date.now(),
    })]
  }
  if (value.subtype === 'session_state_changed') {
    return [frame(sessionId, 'session.updated', {
      running: value.state === 'running' || value.state === 'requires_action',
      waitingOnApproval: value.state === 'requires_action',
      updatedAt: Date.now(),
    })]
  }
  return []
}

function stream(value: Record<string, unknown>, sessionId: string, frame: Frame): EventFrame[] {
  const event = record(value.event)
  if (event === undefined) return []
  switch (event.type) {
    case 'message_start': {
      const message = record(event.message)
      return [frame(sessionId, 'turn.started', { turnId: String(message?.id ?? '') })]
    }
    case 'content_block_start': {
      const block = record(event.content_block)
      if (block?.type !== 'tool_use') return []
      return [frame(sessionId, 'tool.started', {
        callId: String(block.id ?? ''),
        name: String(block.name ?? 'tool'),
        args: record(block.input) ?? {},
      })]
    }
    case 'content_block_delta': {
      const delta = record(event.delta)
      if (delta?.type === 'text_delta' && typeof delta.text === 'string') {
        return [frame(sessionId, 'message.delta', { text: delta.text, itemId: currentMessageId(value) })]
      }
      if (delta?.type === 'thinking_delta' && typeof delta.thinking === 'string') {
        return [frame(sessionId, 'reasoning.delta', { text: delta.thinking, itemId: currentMessageId(value) })]
      }
      return []
    }
    default:
      return []
  }
}

const currentMessageId = (value: Record<string, unknown>): string =>
  typeof value.parent_tool_use_id === 'string' ? value.parent_tool_use_id : 'stream'

function assistant(value: Record<string, unknown>, sessionId: string, frame: Frame): EventFrame[] {
  const message = record(value.message)
  const content = Array.isArray(message?.content) ? message?.content as unknown[] : []
  const events: EventFrame[] = []
  for (const block of content) {
    const item = record(block)
    if (item?.type !== 'text' || typeof item.text !== 'string') continue
    events.push(frame(sessionId, 'message.completed', {
      itemId: String(message?.id ?? ''),
      chars: [...item.text].length,
    }))
  }
  return events
}

function user(value: Record<string, unknown>, sessionId: string, frame: Frame): EventFrame[] {
  const message = record(value.message)
  const content = Array.isArray(message?.content) ? message?.content as unknown[] : []
  const events: EventFrame[] = []
  for (const block of content) {
    const item = record(block)
    if (item?.type !== 'tool_result') continue
    const text = typeof item.content === 'string'
      ? item.content
      : Array.isArray(item.content)
        ? item.content.map(part => typeof record(part)?.text === 'string' ? String(record(part)?.text) : '').join('')
        : ''
    events.push(frame(sessionId, 'tool.completed', {
      callId: String(item.tool_use_id ?? ''),
      status: item.is_error === true ? 'error' : 'completed',
      content: text,
    }))
  }
  return events
}
```

```ts
import type { SettleResult } from '../../adapters/types.ts'
import type { EventFrame } from '../../wire/frames.ts'

export interface ToolContext {
  readonly signal?: AbortSignal
  readonly suggestions?: unknown
  readonly toolUseID?: string
  readonly title?: string
  readonly displayName?: string
  readonly description?: string
}

export type PermissionResult =
  | { behavior: 'allow'; updatedInput: Record<string, unknown>; updatedPermissions?: unknown }
  | { behavior: 'deny'; message: string; interrupt?: boolean }

export interface ElicitationRequest {
  readonly serverName?: string
  readonly message?: string
  readonly requestId?: string
}

export interface ClaudePermissionsOptions {
  readonly timeoutMs: number
  readonly mode: 'remote' | 'notify-only'
  readonly emit: (event: EventFrame) => void
  readonly frame: (sessionId: string, kind: EventFrame['kind'], payload: unknown) => EventFrame
}

export interface ClaudePermissions {
  canUseTool(sessionId: string): (
    toolName: string,
    input: Record<string, unknown>,
    context: ToolContext,
  ) => Promise<PermissionResult>
  onElicitation(sessionId: string): (request: ElicitationRequest) => Promise<unknown>
  settle(sessionId: string, requestId: string, decision: string): SettleResult
  answer(sessionId: string, requestId: string, answers: unknown): SettleResult
  readonly size: number
  dispose(): void
}

interface Pending {
  readonly sessionId: string
  readonly kind: 'approval' | 'question'
  readonly toolKey?: string
  readonly input: Record<string, unknown>
  readonly context: ToolContext
  readonly questions: readonly { id: string; text: string }[]
  readonly resolve: (value: PermissionResult) => void
  readonly timer: NodeJS.Timeout
}

const DECISIONS = ['accept', 'acceptForSession', 'decline', 'cancel'] as const

export function parseSelection(value: unknown, ids: readonly string[]): Record<string, string[]> | undefined {
  if (typeof value !== 'object' || value === null) return undefined
  const raw = (value as { answers?: unknown }).answers
  if (!Array.isArray(raw) || raw.length !== ids.length) return undefined
  const result: Record<string, string[]> = {}
  for (const item of raw) {
    if (typeof item !== 'object' || item === null) return undefined
    const { id, selected } = item as { id?: unknown; selected?: unknown }
    if (typeof id !== 'string' || !ids.includes(id) || result[id] !== undefined) return undefined
    if (!Array.isArray(selected) || !selected.every(label => typeof label === 'string')) return undefined
    result[id] = selected as string[]
  }
  return ids.every(id => result[id] !== undefined) ? result : undefined
}

export function createClaudePermissions(options: ClaudePermissionsOptions): ClaudePermissions {
  const pending = new Map<string, Pending>()
  const byTool = new Map<string, Promise<PermissionResult>>()
  let counter = 0
  const uniqueId = (base: string): string => `${base}-${++counter}`

  const notify = (sessionId: string, requestId: string, card: Record<string, unknown>): void => {
    options.emit(options.frame(sessionId, 'input.requested', {
      requestId, kind: 'approval', agent: 'claude', notifyOnly: true, decisions: [], ...card,
    }))
    options.emit(options.frame(sessionId, 'input.resolved', { requestId, outcome: 'local' }))
  }

  const finish = (
    sessionId: string,
    requestId: string,
    outcome: 'remote' | 'local',
    value: PermissionResult,
  ): SettleResult => {
    const entry = pending.get(requestId)
    if (entry === undefined) return 'unknown'
    if (entry.sessionId !== sessionId) return 'session-mismatch'
    clearTimeout(entry.timer)
    pending.delete(requestId)
    if (entry.toolKey !== undefined) byTool.delete(entry.toolKey)
    options.emit(options.frame(sessionId, 'input.resolved', { requestId, outcome }))
    entry.resolve(value)
    return 'ok'
  }

  return {
    canUseTool(sessionId) {
      return (toolName, input, context) => {
        const toolKey = context.toolUseID === undefined ? undefined : `${sessionId}:${context.toolUseID}`
        if (toolKey !== undefined) {
          const existing = byTool.get(toolKey)
          if (existing !== undefined) return existing
        }

        if (options.mode === 'notify-only') {
          const requestId = uniqueId(`n-${context.toolUseID ?? toolName}`)
          notify(sessionId, requestId, { title: `需要审批：${context.displayName ?? toolName}` })
          return Promise.resolve({ behavior: 'deny', message: 'remote approval disabled' })
        }

        const isQuestion = toolName === 'AskUserQuestion'
        const questions = isQuestion && Array.isArray(input.questions)
          ? (input.questions as unknown[]).map((item, index) => {
            const value = (typeof item === 'object' && item !== null ? item : {}) as Record<string, unknown>
            return {
              id: String(value.id ?? index),
              text: typeof value.question === 'string' ? value.question : String(value.header ?? ''),
              header: value.header,
              options: value.options,
              multiSelect: value.multiSelect === true,
            }
          })
          : []

        const promise = new Promise<PermissionResult>(resolve => {
          const requestId = uniqueId(`${isQuestion ? 'q' : 'p'}-${context.toolUseID ?? toolName}`)
          const timer = setTimeout(() => {
            pending.delete(requestId)
            if (toolKey !== undefined) byTool.delete(toolKey)
            options.emit(options.frame(sessionId, 'input.resolved', { requestId, outcome: 'timeout' }))
            resolve({ behavior: 'deny', message: isQuestion ? 'question-timeout' : 'approval-timeout' })
          }, options.timeoutMs)
          pending.set(requestId, {
            sessionId, kind: isQuestion ? 'question' : 'approval', toolKey, input, context, questions,
            resolve, timer,
          })
          const detail = context.description ?? (typeof input.command === 'string' ? input.command : undefined)
          options.emit(options.frame(sessionId, 'input.requested', {
            requestId,
            kind: isQuestion ? 'question' : 'approval',
            agent: 'claude',
            title: isQuestion ? '需要你的输入' : `需要审批：${context.displayName ?? toolName}`,
            expiresAt: Date.now() + options.timeoutMs,
            ...(detail === undefined ? {} : { detail }),
            ...(isQuestion ? { questions } : { decisions: DECISIONS }),
          }))
        })

        if (toolKey !== undefined) byTool.set(toolKey, promise)
        return promise
      }
    },
    onElicitation(sessionId) {
      return async request => {
        const requestId = uniqueId(`e-${request.serverName ?? 'mcp'}`)
        notify(sessionId, requestId, {
          title: 'MCP 请求（远端只提示）',
          ...(request.message === undefined ? {} : { detail: request.message }),
        })
        return { action: 'decline' }
      }
    },
    settle(sessionId, requestId, decision) {
      const entry = pending.get(requestId)
      if (entry === undefined) return 'unknown'
      if (entry.sessionId !== sessionId) return 'session-mismatch'
      if (entry.kind !== 'approval') return 'invalid'
      if (!(DECISIONS as readonly string[]).includes(decision)) return 'invalid'

      const result: PermissionResult = decision === 'accept'
        ? { behavior: 'allow', updatedInput: entry.input }
        : decision === 'acceptForSession'
          ? {
              behavior: 'allow',
              updatedInput: entry.input,
              ...(entry.context.suggestions === undefined ? {} : { updatedPermissions: entry.context.suggestions }),
            }
          : decision === 'cancel'
            ? { behavior: 'deny', message: 'cancelled by remote operator', interrupt: true }
            : { behavior: 'deny', message: 'declined by remote operator' }
      return finish(sessionId, requestId, 'remote', result)
    },
    answer(sessionId, requestId, answers) {
      const entry = pending.get(requestId)
      if (entry === undefined) return 'unknown'
      if (entry.sessionId !== sessionId) return 'session-mismatch'
      if (entry.kind !== 'question') return 'invalid'
      const ids = entry.questions.map(question => question.id)
      const selected = parseSelection(answers, ids)
      if (selected === undefined) return 'invalid'

      const mapped: Record<string, string> = {}
      for (const question of entry.questions) mapped[question.text] = (selected[question.id] ?? []).join(', ')
      return finish(sessionId, requestId, 'remote', {
        behavior: 'allow',
        updatedInput: { ...entry.input, answers: mapped },
      })
    },
    get size() { return pending.size },
    dispose() {
      for (const [requestId, entry] of pending) {
        clearTimeout(entry.timer)
        pending.delete(requestId)
        entry.resolve({ behavior: 'deny', message: 'disposed' })
      }
      byTool.clear()
    },
  }
}
```

```ts
export interface FakeQuery extends ClaudeQueryLike {
  readonly calls: string[]
  readonly inputs: unknown[]
  readonly pushed: unknown[]
  closed: boolean
  push(message: unknown): void
}

/** 假 runtime 只认 sessionId `s1`；init 里的 session_id 用它，测试断言才好写。 */
export function createFakeClaude(options: {
  sessions?: readonly ClaudeSessionInfo[]
  models?: readonly ClaudeModelInfo[]
} = {}) {
  const queries: FakeQuery[] = []
  const runtime: ClaudeRuntime = {
    query(input) {
      const queue: unknown[] = []
      let wake: (() => void) | undefined
      let ended = false
      const resume = (): void => { const fn = wake; wake = undefined; fn?.() }

      const query: FakeQuery = {
        calls: [],
        inputs: [],
        pushed: [],
        closed: false,
        push(message) {
          queue.push(message)
          query.pushed.push(message)
          resume()
        },
        async interrupt() { query.calls.push('interrupt') },
        async setPermissionMode(mode) {
          query.calls.push(`permission:${mode}`)
          query.push({ type: 'system', subtype: 'init', session_id: 's1', model: 'm1', permissionMode: mode })
        },
        async setModel(model) {
          query.calls.push(`model:${model}`)
          query.push({ type: 'system', subtype: 'init', session_id: 's1', model, permissionMode: 'default' })
        },
        close() {
          query.closed = true
          ended = true
          resume()
        },
        async *[Symbol.asyncIterator]() {
          while (true) {
            const next = queue.shift()
            if (next !== undefined) { yield next; continue }
            if (ended) return
            await new Promise<void>(resolve => { wake = resolve })
          }
        },
      }

      queries.push(query)
      void (async () => { for await (const item of input.prompt) query.inputs.push(item) })()
      return query
    },
    listSessions: async () => options.sessions ?? [],
    getSessionMessages: async () => [],
    supportedModels: async () => options.models ?? [],
  }
  return { runtime, queries }
}
```

```ts
import type { AgentAdapter, AgentProbe } from '../types.ts'
import type { Capability, Tunable } from '../../adapters/types.ts'
import type { EventFrame } from '../../wire/frames.ts'
import { isTightening } from '../../tunables/tighten.ts'
import { createEventQueue, type EventQueue } from '../queue.ts'
import { createFrameFactory } from '../frames.ts'
import { listClaudeSessions, toSessionEvents } from './discover.ts'
import { toMarshalEvents } from './map.ts'
import { createClaudePermissions, type ClaudePermissions } from './permissions.ts'
import type { ClaudeModelInfo, ClaudeQueryLike, ClaudeRuntime } from './runtime.ts'

export interface ClaudeAdapterOptions {
  readonly device: string
  readonly runtime: ClaudeRuntime
  readonly emit: (event: EventFrame) => void
  readonly approvalMode: 'remote' | 'notify-only' | 'local-only'
  readonly approvalTimeoutMs: number
  readonly idleMs?: number
  readonly readbackTimeoutMs?: number
  readonly executablePath?: string
  readonly probe?: AgentProbe
}

const PERMISSION_VALUES = [
  { id: 'bypassPermissions', label: '绕过权限' },
  { id: 'acceptEdits', label: '自动接受编辑' },
  { id: 'auto', label: '自动' },
  { id: 'default', label: '默认' },
  { id: 'dontAsk', label: '不再询问' },
  { id: 'plan', label: '计划模式' },
]

interface Session {
  readonly sessionId: string
  query?: ClaudeQueryLike
  input?: EventQueue<unknown>
  readonly events: EventQueue<EventFrame>
  settings: { model?: string; permission?: string; effort?: string }
  appliedEffort?: string
  initSeq: number
  init?: Record<string, unknown>
  waiters: Set<() => void>
  lastActivity: number
  timer?: NodeJS.Timeout
  generation: number
  closed: boolean
}

const message = (error: unknown): string => error instanceof Error ? error.message : String(error)

const isInit = (value: unknown): value is Record<string, unknown> => {
  if (typeof value !== 'object' || value === null) return false
  const record = value as Record<string, unknown>
  return record.type === 'system' && record.subtype === 'init'
}

export function createClaudeAdapter(options: ClaudeAdapterOptions): AgentAdapter {
  const frame = createFrameFactory(options.device, 'claude')
  const sessions = new Map<string, Session>()
  const permissions: ClaudePermissions = createClaudePermissions({
    mode: options.approvalMode === 'notify-only' ? 'notify-only' : 'remote',
    timeoutMs: options.approvalTimeoutMs,
    emit: options.emit,
    frame,
  })
  const modelCatalog: ClaudeModelInfo[] = []
  let lastProbe: AgentProbe = options.probe ?? { available: true, missing: [] }

  const waitForInit = (session: Session, afterSeq: number): Promise<Record<string, unknown> | undefined> =>
    new Promise(resolve => {
      if (session.initSeq > afterSeq) { resolve(session.init); return }
      const timer = setTimeout(() => {
        session.waiters.delete(done)
        resolve(undefined)
      }, options.readbackTimeoutMs ?? 1_500)
      const done = (): void => { clearTimeout(timer); resolve(session.init) }
      session.waiters.add(done)
    })

  const emitCapability = (sessionId: string): void => {
    options.emit(frame(sessionId, 'capability.changed', {
      tunables: tunables(sessionId), readbackAt: Date.now(),
    }))
  }

  const armIdle = (session: Session): void => {
    if (session.timer !== undefined) clearTimeout(session.timer)
    const idleMs = options.idleMs ?? 10 * 60_000
    session.timer = setTimeout(() => {
      if (Date.now() - session.lastActivity < idleMs) { armIdle(session); return }
      closeSession(session.sessionId, 'idle')
    }, idleMs)
    session.timer.unref()
  }

  const pump = async (session: Session, query: ClaudeQueryLike, generation: number): Promise<void> => {
    try {
      for await (const item of query) {
        if (session.generation !== generation || session.closed) return
        session.lastActivity = Date.now()
        if (isInit(item)) {
          session.init = item
          session.initSeq += 1
          if (typeof item.model === 'string') session.settings.model = item.model
          if (typeof item.permissionMode === 'string') session.settings.permission = item.permissionMode
          for (const waiter of session.waiters) waiter()
          session.waiters.clear()
        }
        for (const event of toMarshalEvents(item, session.sessionId, frame)) session.events.push(event)
      }
    } catch (error) {
      if (session.generation === generation && !session.closed) {
        session.events.push(frame(session.sessionId, 'error', {
          code: 'adapter-crashed', message: message(error),
        }))
      }
    }
  }

  const openQuery = (session: Session): void => {
    const input = createEventQueue<unknown>()
    const generation = session.generation + 1
    session.generation = generation
    session.input = input
    session.appliedEffort = session.settings.effort
    const query = options.runtime.query({
      prompt: input,
      options: {
        resume: session.sessionId,
        includePartialMessages: true,
        ...(session.settings.model === undefined ? {} : { model: session.settings.model }),
        ...(session.settings.permission === undefined ? {} : { permissionMode: session.settings.permission }),
        ...(session.settings.effort === undefined ? {} : { effort: session.settings.effort }),
        ...(options.executablePath === undefined ? {} : { pathToClaudeCodeExecutable: options.executablePath }),
        ...(options.approvalMode === 'local-only'
          ? {}
          : {
              canUseTool: permissions.canUseTool(session.sessionId),
              onElicitation: permissions.onElicitation(session.sessionId),
            }),
      },
    })
    session.query = query
    void pump(session, query, generation)
  }

  const closeSession = (sessionId: string, reason: string): void => {
    const session = sessions.get(sessionId)
    if (session === undefined) return
    session.closed = true
    if (session.timer !== undefined) clearTimeout(session.timer)
    session.input?.close()
    session.query?.close()
    session.events.push(frame(sessionId, 'session.updated', {
      ownership: 'foreign-idle', running: false, updatedAt: Date.now(), reason,
    }))
    session.events.close()
    sessions.delete(sessionId)
    emitCapability(sessionId)
  }

  const ensure = (sessionId: string): Session => {
    const existing = sessions.get(sessionId)
    if (existing !== undefined) return existing
    const session: Session = {
      sessionId,
      events: createEventQueue(),
      settings: {},
      initSeq: 0,
      waiters: new Set(),
      lastActivity: Date.now(),
      generation: 0,
      closed: false,
    }
    sessions.set(sessionId, session)
    openQuery(session)
    session.events.push(frame(sessionId, 'session.updated', {
      ownership: 'ours', running: true, updatedAt: Date.now(),
    }))
    armIdle(session)
    return session
  }

  const tunables = (sessionId: string): readonly Tunable[] => {
    const session = sessions.get(sessionId)
    const settings = session?.settings ?? {}
    const model = modelCatalog.find(entry => entry.value === settings.model)
    const efforts = model?.supportedEffortLevels ?? []
    return [
      {
        id: 'model', scope: 'next-turn', effect: 'immediate', editable: true,
        values: modelCatalog.map(entry => ({ id: entry.value, label: entry.displayName ?? entry.value })),
        ...(settings.model === undefined ? {} : { current: settings.model }),
      },
      {
        id: 'reasoning', scope: 'next-turn', effect: 'immediate', editable: true, direction: 'tighten-only',
        values: efforts.map(effort => ({ id: effort, label: effort })),
        ...(settings.effort === undefined ? {} : { current: settings.effort }),
      },
      {
        id: 'permission', scope: 'next-turn', effect: 'immediate', editable: true, direction: 'tighten-only',
        values: PERMISSION_VALUES.map(value => ({ ...value })),
        ...(settings.permission === undefined ? {} : { current: settings.permission }),
      },
      {
        id: 'plan', scope: 'next-turn', effect: 'immediate', editable: true,
        values: [{ id: 'on', label: '计划模式' }, { id: 'off', label: '普通模式' }],
        current: settings.permission === 'plan' ? 'on' : 'off',
      },
      { id: 'goal', scope: 'next-turn', effect: 'immediate', editable: false, reason: 'Claude 没有独立的目标状态' },
      { id: 'skill', scope: 'host', effect: 'needs-restart', editable: false, reason: '技能是宿主级配置，远端只能查看' },
      { id: 'plugin', scope: 'host', effect: 'needs-restart', editable: false, reason: '插件是宿主级配置，远端只能查看' },
    ]
  }

  const configure = async (sessionId: string, args: Record<string, unknown>): Promise<unknown> => {
    const field = args.field as Tunable['id'] | undefined
    if (field === undefined) throw new Error('invalid-args: field is required')
    const tunable = tunables(sessionId).find(item => item.id === field)
    if (tunable === undefined) throw new Error(`unsupported: ${field}`)
    if (!tunable.editable) throw new Error(`not-editable: ${field}: ${tunable.reason ?? ''}`)

    const session = ensure(sessionId)
    if (field === 'permission' || field === 'reasoning') {
      const from = tunable.current
      if (from === undefined) throw new Error(`readback-timeout: ${field}`)
      if (from !== args.value && !isTightening({
        agent: 'claude', field, from, to: args.value, values: tunable.values,
      })) {
        throw new Error(`widen-rejected: ${String(from)} -> ${String(args.value)}`)
      }
    }

    if (field === 'reasoning') {
      session.settings.effort = String(args.value)
      emitCapability(sessionId)
      return { settings: session.settings, effective: 'next-turn' }
    }

    if (field === 'plan') {
      const desired = args.value === true || args.value === 'on' || args.value === 'plan' ? 'plan' : 'default'
      if (desired === 'default' && session.settings.permission === 'plan') {
        throw new Error('widen-rejected: plan -> default')
      }
      const before = session.initSeq
      await session.query?.setPermissionMode(desired)
      session.settings.permission = desired
      const init = await waitForInit(session, before)
      if (init !== undefined && typeof init.permissionMode === 'string' && init.permissionMode !== desired) {
        throw new Error(`readback-mismatch: plan ${String(init.permissionMode)} != ${desired}`)
      }
      emitCapability(sessionId)
      return { settings: session.settings, effective: init === undefined ? 'next-turn' : 'immediate' }
    }

    if (field === 'model') {
      const desired = args.value
      if (typeof desired !== 'string') throw new Error('invalid-args: model requires a string')
      if (modelCatalog.length > 0 && !modelCatalog.some(entry => entry.value === desired)) {
        throw new Error(`invalid-args: unknown model ${desired}`)
      }
      const before = session.initSeq
      await session.query?.setModel(desired)
      session.settings.model = desired
      const init = await waitForInit(session, before)
      if (init !== undefined && typeof init.model === 'string' && init.model !== desired) {
        throw new Error(`readback-mismatch: model ${String(init.model)} != ${desired}`)
      }
      emitCapability(sessionId)
      return { settings: session.settings, effective: init === undefined ? 'next-turn' : 'immediate' }
    }

    if (field === 'permission') {
      const desired = String(args.value)
      const before = session.initSeq
      await session.query?.setPermissionMode(desired)
      session.settings.permission = desired
      const init = await waitForInit(session, before)
      if (init !== undefined && typeof init.permissionMode === 'string' && init.permissionMode !== desired) {
        throw new Error(`readback-mismatch: permission ${String(init.permissionMode)} != ${desired}`)
      }
      emitCapability(sessionId)
      return { settings: session.settings, effective: init === undefined ? 'next-turn' : 'immediate' }
    }

    throw new Error(`unsupported: ${field}`)
  }

  return {
    agent: 'claude',
    get lastProbe() { return lastProbe },
    capabilities(sessionId) {
      const capability: Capability = {
        observe: 'stream',
        dispatch: 'prompt',
        reasoning: true,
        toolDetail: 'args+output',
        promptFidelity: 'native',
        tunables: tunables(sessionId),
      }
      return capability
    },
    frame,
    tunables,
    ownership(sessionId) { return sessions.has(sessionId) ? 'ours' : 'foreign-idle' },
    async probe() { return lastProbe },
    async start() {
      try {
        const models = await options.runtime.supportedModels()
        modelCatalog.splice(0, modelCatalog.length, ...models)
        lastProbe = { available: true, missing: [] }
      } catch (error) {
        lastProbe = { available: false, missing: ['claude'], reason: message(error) }
      }
    },
    async stop() {
      for (const sessionId of [...sessions.keys()]) closeSession(sessionId, 'stop')
      permissions.dispose()
    },
    async snapshot(signal) {
      const list = await listClaudeSessions(options.runtime)
      if (signal.aborted) return []
      const events = toSessionEvents(list, frame, sessionId => sessions.has(sessionId))
      for (const session of list) {
        events.push(frame(session.sessionId, 'capability.changed', {
          tunables: tunables(session.sessionId), readbackAt: Date.now(),
        }))
      }
      return events
    },
    async *follow(sessionId, signal) {
      const session = ensure(sessionId)
      const onAbort = (): void => { /* 只停止消费，session 生命周期由 idle 与 stop 管理 */ }
      signal.addEventListener('abort', onAbort, { once: true })
      try {
        yield* session.events
      } finally {
        signal.removeEventListener('abort', onAbort)
      }
    },
    async execute(command) {
      const sessionId = command.target.sessionId
      const args = (command.args ?? {}) as Record<string, unknown>
      switch (command.op) {
        case 'prompt': {
          if (typeof args.text !== 'string' || args.text.trim() === '') {
            throw new Error('invalid-args: text is required')
          }
          const session = ensure(sessionId)
          if (session.settings.effort !== undefined && session.settings.effort !== session.appliedEffort) {
            session.input?.close()
            session.query?.close()
            openQuery(session)
          }
          session.lastActivity = Date.now()
          armIdle(session)
          session.input?.push({
            type: 'user',
            session_id: sessionId,
            parent_tool_use_id: null,
            message: { role: 'user', content: [{ type: 'text', text: args.text }] },
          })
          return { accepted: true }
        }
        case 'interrupt': {
          const session = sessions.get(sessionId)
          if (session === undefined) throw new Error('session-unknown: not resumed')
          await session.query?.interrupt()
          return { interrupted: true }
        }
        case 'decide': {
          if (typeof args.requestId !== 'string' || typeof args.decision !== 'string') {
            throw new Error('invalid-args: requestId and decision are required')
          }
          const outcome = permissions.settle(sessionId, args.requestId, args.decision)
          if (outcome !== 'ok') throw new Error(`${outcome}: ${args.requestId}`)
          return { accepted: true }
        }
        case 'answer': {
          if (typeof args.requestId !== 'string' || args.answers === undefined) {
            throw new Error('invalid-args: requestId and answers are required')
          }
          const outcome = permissions.answer(sessionId, args.requestId, args.answers)
          if (outcome !== 'ok') throw new Error(`${outcome}: ${args.requestId}`)
          return { accepted: true }
        }
        case 'configure':
          return configure(sessionId, args)
        default:
          throw new Error(`unsupported: ${command.op}`)
      }
    },
  }
}
```

说明：`follow` 只消费 `session.events` 这一个队列，所以同一会话只能有一个 follower；node 的 runner 保证这一点。`interrupt` 在未续跑时返回 `session-unknown`，与 DSH 适配器一致。

```ts
function parseApprovalRelay(raw: unknown): ApprovalRelay {
  const value = raw ?? 'off'
  if (value !== 'off' && value !== 'notify' && value !== 'remote') {
    throw new Error('marshal: config.approvalRelay must be "off", "notify" or "remote"')
  }
  return value
}

function parseAgents(raw: unknown): readonly AgentKind[] {
  const value = raw ?? ['dsh']
  if (!Array.isArray(value) || value.length === 0) throw new Error('marshal: config.agents must be a non-empty array')
  for (const agent of value) {
    if (!AGENT_KINDS.includes(agent as AgentKind)) {
      throw new Error(`marshal: config.agents contains unknown agent ${String(agent)}`)
    }
  }
  return [...new Set(value as AgentKind[])]
}

function parsePositive(raw: unknown, key: string, fallback: number): number {
  const value = raw ?? fallback
  if (typeof value !== 'number' || !Number.isFinite(value) || value <= 0) {
    throw new Error(`marshal: config.${key} must be a positive number`)
  }
  return value
}
```

```ts
export interface CodexCommand {

  readonly args: readonly string[]

}

/** 显式路径优先；否则从已安装的 `@openai/codex` 包读 `bin.codex`，不猜任何绝对路径。 */
export function resolveCodexCommand(executable?: string): CodexCommand {
  if (executable !== undefined && executable.trim() !== '') {
    return { command: executable, args: ['app-server', '--stdio'] }
  }
  try {
    const require = createRequire(import.meta.url)
    const pkgPath = require.resolve('@openai/codex/package.json')
    const pkg = JSON.parse(readFileSync(pkgPath, 'utf8')) as {
      version?: string
      bin?: string | Record<string, string>
    }

      ...(pkg.version === undefined ? {} : { version: pkg.version }),
    }
  } catch (error) {
    throw new Error(`agent-missing: codex (${error instanceof Error ? error.message : String(error)})`)
  }
```

```ts
import { encodeFrame, type EventFrame, type Frame } from '../wire/frames.ts'
import type { AgentHost } from '../agents/types.ts'
import type { LoggerLike } from '../dsh-types.ts'
import type { NodeLink } from './link.ts'

export interface Followers {
  start(agent: string, sessionId: string): Promise<void>
  stopAll(): void
}

export function createFollowers(options: {
  host: AgentHost
  send: (event: EventFrame) => void
  logger: Pick<LoggerLike, 'warn'>
}): Followers {
  const active = new Map<string, AbortController>()
  return {
    async start(agent, sessionId) {
      const key = `${agent}\u0000${sessionId}`
      if (active.has(key)) return
      const abort = new AbortController()
      const iterator = options.host.follow(agent, sessionId, abort.signal)[Symbol.asyncIterator]()
      const first = await iterator.next()
      if (first.done === true) throw new Error('resume-failed: no events')
      active.set(key, abort)
      options.send(first.value)
      void (async () => {
        try {
          for await (const event of { [Symbol.asyncIterator]: () => iterator }) options.send(event)
        } catch (error) {
          if (!abort.signal.aborted) {
            options.logger.warn(`marshal: follow ${agent}/${sessionId} ended: ${String(error)}`)
          }
        } finally {
          active.delete(key)
        }
      })()
    },
    stopAll() {
      for (const abort of active.values()) abort.abort()
      active.clear()
    },
  }
}

/** 错误前缀即 wire 错误码：`widen-rejected: ...` → `widen-rejected`。 */
function codeOf(error: unknown): string {
  const text = error instanceof Error ? error.message : String(error)
  return /^([a-z][a-z-]*):/.exec(text)?.[1] ?? 'adapter-crashed'
}

export function runCommand(
  link: NodeLink,
  host: AgentHost,
  followers: Followers,
  frame: Frame,
  device: string,
): void {
  if (frame.type !== 'command') return
  const reply = (body: { ok: boolean; result?: unknown; code?: string; message?: string }): void => {
    link.sendRaw(encodeFrame({ v: 1, type: 'ack', id: frame.id, ...body }))
  }
  if (frame.target.device !== device) { reply({ ok: false, code: 'device-mismatch' }); return }

  if (frame.op === 'resume') {
    void followers.start(frame.target.agent, frame.target.sessionId)
      .then(() => reply({ ok: true, result: { following: true } }))
      .catch((error: unknown) => reply({
        ok: false,
        code: codeOf(error),
        message: error instanceof Error ? error.message : String(error),
      }))
    return
  }

  void host.execute({ op: frame.op, target: frame.target, args: frame.args })
    .then(result => reply({ ok: true, result }))
    .catch((error: unknown) => reply({
      ok: false,
      code: codeOf(error),
      message: error instanceof Error ? error.message : String(error),
    }))
}
```

`resume` 的失败现在能回给 hub：`follow` 的第一条事件前必须完成续跑（Codex 的 `thread/resume`、Claude 的 `Query` 初始化），所以 `iterator.next()` 会带上 `resume-failed` / `writer-busy`。

```ts
  if (controller === undefined) throw new Error('marshal: node role requires the sessionController service')

  const patterns = config.redaction.map(source => new RegExp(source, 'gi'))
  const pipeline = (event: EventFrame): EventFrame => {
    const masked = config.privacy === 'content' ? event : stripContent(event)
    return { ...masked, payload: redact(masked.payload, patterns) }
  }

  let link!: NodeLink
  const emit = (event: EventFrame): void => { link.sendEvent(pipeline(event)) }

  let relay: QuestionRelay | undefined
  const dsh = config.agents.includes('dsh')
    ? createDshAdapter(createDshSurface(controller), {
        device: config.deviceName!,
        answers: {
          settle: (sessionId, requestId, answers) => relay?.settle(sessionId, requestId, answers) ?? 'unknown',
        },
      })
    : undefined
  relay = dsh !== undefined && config.questionRelay === 'remote'
    ? createQuestionRelay({ ctx, timeoutMs: config.answerTimeoutMs, send: emit, frame: dsh.frame })
    : undefined

  const builders: AgentBuilders = {

      })

        close: () => { void supervisor.stop() },

        device: config.deviceName!,
        transport,

        onPeer: created => { peer = created },
      })
      return wrapWithSupervisor(adapter, supervisor)
    },
    claude: async () => {
      const runtime = await createSdkRuntime({
        ...(config.claudeExecutable === undefined ? {} : { executablePath: config.claudeExecutable }),
      })
      return createClaudeAdapter({
        device: config.deviceName!,
        runtime,
        emit,
        approvalMode: config.approvalRelay === 'off'
          ? 'local-only'
          : config.approvalRelay === 'notify' ? 'notify-only' : 'remote',
        approvalTimeoutMs: config.approvalTimeoutMs,
        idleMs: 10 * 60_000,
      })
    },
```

`src/hub/buffer.ts`：

```ts
import type { EventFrame } from '../wire/frames.ts'

export interface EventBuffer {
  append(event: EventFrame): void
  since(afterMs: number, device?: string): EventFrame[]
  size(): number
}

interface Entry {
  readonly at: number
  readonly event: EventFrame
}

export function createEventBuffer(options: {
  now?: () => number
  windowMs?: number
  perSessionLimit?: number
} = {}): EventBuffer {
  const now = options.now ?? Date.now
  const windowMs = options.windowMs ?? 15 * 60_000
  const perSessionLimit = options.perSessionLimit ?? 500
  const entries: Entry[] = []
  const bySession = new Map<string, Entry[]>()

  const keyOf = (event: EventFrame): string =>
    `${event.device}\u0000${event.agent}\u0000${event.sessionId}`

  const drop = (entry: Entry): void => {
    const index = entries.indexOf(entry)
    if (index >= 0) entries.splice(index, 1)
    const key = keyOf(entry.event)
    const list = bySession.get(key)
    if (list === undefined) return
    const position = list.indexOf(entry)
    if (position >= 0) list.splice(position, 1)
    if (list.length === 0) bySession.delete(key)
  }

  const evict = (at: number): void => {
    while (entries.length > 0 && at - entries[0]!.at > windowMs) drop(entries[0]!)
  }

  return {
    append(event) {
      const at = now()
      evict(at)
      const entry: Entry = { at, event }
      entries.push(entry)
      const key = keyOf(event)
      const list = bySession.get(key) ?? []
      list.push(entry)
      bySession.set(key, list)
      while (list.length > perSessionLimit) drop(list[0]!)
    },
    since(afterMs, device) {
      return entries
        .filter(entry => entry.at > afterMs && (device === undefined || entry.event.device === device))
        .map(entry => entry.event)
    },
    size() { return entries.length },
  }
}
```

```ts
function scriptedTransport() {
  const written: { id?: number; method?: string; result?: unknown }[] = []
  let feed: ((line: string) => void) | undefined

  const transport = {
    write(text: string) {
      for (const line of text.trim().split('\n')) {
        const message = JSON.parse(line) as { id?: number; method?: string }
        written.push(message)
        if (message.id === undefined) continue
        const reply = (result: unknown): void => { feed?.(JSON.stringify({ id: message.id, result })) }
        switch (message.method) {
          case 'initialize': reply({}); break
          case 'model/list':
            reply({ data: [{ id: 'm1', model: 'm1', displayName: 'M1',
              supportedReasoningEfforts: [{ reasoningEffort: 'low' }, { reasoningEffort: 'high' }] }] })
            break
          case 'thread/list': reply({ data: [thread] }); break
          case 'thread/resume':
            reply({ thread: { id: 't1' }, threadSettings: {
              model: 'm1', effort: 'high', sandboxPolicy: { type: 'workspaceWrite' },
              collaborationMode: { mode: 'default' },
            } })
            break
          case 'turn/start': reply({ turn: { id: 'u1' } }); break
          default: reply({}); break
        }
      }
    },
    close() {},
  }

  return {
    transport,
    written,
    feedLine: (line: string): void => { feed?.(line) },
    setFeed: (fn: (line: string) => void): void => { feed = fn },
  }
}
```
